-- Genera un sistema de gestion de ventas donde cada pedido pertenece a un unico cliente pero un 
-- cliente puede tener varios pedidos a lo largo del tiempo.
-- Relacion 1:m (cliente,pedidos)

create database gestion_de_ventas;
use gestion_de_ventas;

create table clientes (
id int primary key auto_increment,
cliente varchar(100)
);

insert into clientes (cliente) values
("Anton"),
("Jesus");

create table pedidos (
id int primary key auto_increment,
clientes_id int,
num_pedido varchar(100),
cantidad int,
foreign key (clientes_id) references clientes(id)
);

insert into pedidos (clientes_id,num_pedido, cantidad) values
(1,1,250000),
(1,2,200000),
(2,1,300000);

select * from pedidos;
