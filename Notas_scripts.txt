# Seleccionamos una tabla
SELECT * FROM salaries;
# Una vez seleccionada la tabla utilizamos WHERE
SELECT * FROM salaries Where emp_no = 10002;
SELECT * FROM salaries Where salary = 67534;

# Operadores de comparacion <,>,!=,=
SELECT 4 < 5; # 1 == verdadero
SELECT 4 > 5; # 1 == falso
SELECT * FROM salaries Where salary = 60117;
SELECT * FROM salaries Where salary != 60117;

# Operadores logicos (AND, OR, NOT)
SELECT TRUE AND TRUE;
SELECT TRUE AND FALSE;
SELECT 1 AND 1;

SELECT * FROM employees;
# Desplazamiento horizontal (WHERE)
SELECT * FROM employees WHERE gender = "M" AND last_name = "Facello";
# Que cambia si ahora utilizamos el operador OR ?
SELECT * FROM employees WHERE gender = "M" OR last_name = "Facello";

# ¿ Como poder utilizar el operador NOT ?
SELECT NOT False;
# Muestra todos aquellos empleados que sean femeninos
SELECT * FROM employees WHERE gender != "M";
# O bien, usando el operador NOT
SELECT * FROM employees WHERE NOT gender = "M";

# Between: 
# Sirve para realizar consultas que tengan un rango 
SELECT * FROM employees WHERE emp_no BETWEEN 10001 AND 10005;
# ¿Como podria ejecutar la linea anterior sin utilizar el operador BETWEEN?
SELECT * FROM employees WHERE emp_no >= 10001 AND emp_no <= 10005 ;
# Considerando que ahora NO queremos mostrar todos aquellos empleados que se encuentran entre
# 10001 y 10005
SELECT * FROM employees WHERE emp_no NOT BETWEEN 10001 AND 10005;
# Aqui es importante considerar el orden de los elementos para BETWEEN
SELECT * FROM employees WHERE emp_no BETWEEN 10005 AND 10001; # ERROR

# Apliquemos BETWEEN para el caso de los cumpleaños en el año 53
SELECT * FROM employees WHERE birth_date BETWEEN "1953-01-01" AND "1953-12-31";
