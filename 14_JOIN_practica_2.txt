-- Qué ciudades cuentan con estaciones meteorológicas.

-- Qué estaciones sí tienen mediciones registradas.

-- Qué ciudades no tienen estaciones.
SELECT * FROM ciudades S JOIN estaciones T ON S.id_ciudad = T.id_ciudad;
