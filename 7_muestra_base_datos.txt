CREATE DATABASE ciencia_datos_base;
USE ciencia_datos_base;

-- Tabla de alumnos:
CREATE TABLE alumnos (
	id INT PRIMARY KEY AUTO_INCREMENT,
	nombre VARCHAR(100),
    correo VARCHAR(100),
    turno VARCHAR(100),
    carrera VARCHAR(100),
    sexo VARCHAR(100)
);
